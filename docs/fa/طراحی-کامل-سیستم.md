### ۱.۲ مشخصات فنی سرور
• CPU: 8 cores minimum (برای پردازش تصویر)
• RAM: 32GB minimum
• GPU: NVIDIA T4 minimum (برای OCR و پردازش تصویر)
• Storage: 1TB SSD + 5TB HDD (برای ذخیره نقشه‌ها)
• Bandwidth: 10TB monthly
• Backup: روزانه + real-time replication

## ۲. منطق هوش مصنوعی (Decision Engine)

### ۲.۱ ساختار درختی تصمیم‌گیری

\\\
تعمیرات موبایل (مثال کامل):
├─ دستگاه روشن نمی‌شود
│  ├─ پاور باتری تست شود (دستور: مولتی‌متر روی پین‌های باتری)
│  │  ├─ ولتاژ < 3.3V → تعویض باتری (تصویر: محل باتری + روش تعویض)
│  │  └─ ولتاژ نرمال → بررسی پاور IC
│  │       ├─ عکس از پاور IC بگیرید
│  │       ├─ OCR خواندن شماره IC
│  │       ├─ نقشه: پین‌های VIN, VOUT, EN
│  │       ├─ تست ولتاژ پین EN
│  │       │   ├─ EN = LOW → بررسی PMIC
│  │       │   └─ EN = HIGH → پاور IC معیوب
│  │       └─ دستور: تعویض پاور IC (ویدئو: حذف و نصب)
├─ تصویر ندارد اما صدا دارد
│  ├─ بررسی Backlight
│  │   ├─ فلش‌لایت به صفحه بتابانید
│  │   │   ├─ تصویر دیده می‌شود → تعویض LED Backlight
│  │   │   └─ تصویر دیده نمی‌شود → بررسی مبدل
│  │   └─ ...
└─ آنتن دهی ضعیف
    ├─ تست آنتن با Spectrum Analyzer (دستور: اتصال)
    ├─ ...

### ۲.۲ قوانین موتور تصمیم‌گیری

\\\javascript
class DecisionEngine {
  constructor() {
    this.currentStep = null;
    this.history = [];
    this.lockedSteps = new Set();
  }

  // قفل کردن مرحله - غیرقابل بازگشت بدون تأیید
  lockStep(stepId, reason) {
    this.lockedSteps.add(stepId);
    this.log(\Step \ locked: \\);
  }

  // بررسی اجازه حرکت به مرحله بعد
  canProceed(currentStep, nextStep) {
    if (this.lockedSteps.has(currentStep)) {
      return {
        allowed: false,
        reason: "این مرحله خطرناک است و باید تکمیل شود",
        requiredAction: "تأیید با رمز امنیتی"
      };
    }
    
    // بررسی وابستگی‌ها
    const dependencies = this.getDependencies(nextStep);
    const missing = dependencies.filter(d => !this.history.includes(d));
    
    if (missing.length > 0) {
      return {
        allowed: false,
        reason: \مراحل \必须先完成\,
        requiredAction: "بازگشت و تکمیل مراحل لازم"
      };
    }
    
    return { allowed: true };
  }
}

### ۳.۱ دوربین داخلی اپلیکیشن

\\\javascript
class AdvancedCamera {
  constructor() {
    this.modes = {
      NORMAL: 'normal',
      MACRO: 'macro',          // لوپ دیجیتال
      MANUAL_FOCUS: 'manual',
      BOARD_SCAN: 'board_scan'
    };
    
    this.currentMode = this.modes.NORMAL;
    this.zoomLevel = 1.0;
    this.focusLocked = false;
  }

  // فعال‌سازی لوپ دیجیتال (10x zoom دیجیتال)
  enableDigitalLoop() {
    this.currentMode = this.modes.MACRO;
    this.zoomLevel = 10.0;
    this.applyFilters(['sharpen', 'contrast', 'edge_enhance']);
    return this;
  }

  // قفل فوکوس روی نقطه خاص
  lockFocus(x, y) {
    this.focusLocked = true;
    this.focusPoint = { x, y };
    this.adjustFocusAlgorithm();
  }

  // الگوریتم تشخیص برد
  detectBoard(image) {
    const steps = [
      'convert_to_grayscale',
      'apply_gaussian_blur',
      'canny_edge_detection',
      'find_contours',
      'filter_quadrilaterals',
      'perspective_transform'
    ];
    
    return this.processPipeline(image, steps);
  }
}

## ۵. پایگاه داده

### ۵.۱ طراحی جداول اصلی

\\\sql
-- جدول کاربران
CREATE TABLE users (
  user_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  phone_number VARCHAR(15) UNIQUE NOT NULL,
  email VARCHAR(255) UNIQUE,
  full_name VARCHAR(100),
  hashed_password VARCHAR(255) NOT NULL,
  is_active BOOLEAN DEFAULT true,
  registration_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  last_login TIMESTAMP
);

-- جدول اشتراک‌ها
CREATE TABLE subscriptions (
  subscription_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
  category VARCHAR(50) NOT NULL,
  plan_type VARCHAR(20) NOT NULL,
  amount DECIMAL(10, 2) NOT NULL,
  start_date TIMESTAMP NOT NULL,
  end_date TIMESTAMP NOT NULL,
  is_active BOOLEAN DEFAULT true
);

-- جدول تعمیرات
CREATE TABLE repair_sessions (
  session_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(user_id),
  device_model VARCHAR(100) NOT NULL,
  problem_description TEXT NOT NULL,
  status VARCHAR(20) DEFAULT 'in_progress',
  started_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  completed_at TIMESTAMP,
  success_rating INTEGER
);
